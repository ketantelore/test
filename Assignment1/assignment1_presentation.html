def count_inversions(state):
    flat_list = []
    # Flatten 2D to 1D list, ignoring the blank (0)
    for row in range(3):
        for col in range(3):
            tile = state[row][col]
            if tile != 0:
                flat_list.append(tile)

    inversions = 0
    total_tiles = len(flat_list)  # Should be 8 (since we removed 0)
    for i in range(total_tiles):
        for j in range(i + 1, total_tiles):
            if flat_list[i] > flat_list[j]:
                inversions += 1
    return inversions

def is_solvable(state):
    total_inversions = count_inversions(state)
    print(f"Total inversions found: {total_inversions}")
    if total_inversions % 2 == 0:
        return True
    else:
        return False

def change_state(state, dir):
    blank_row = -1
    blank_col = -1
    
    for row in range(3):
        for col in range(3):

                blank_row = row
                blank_col = col
                
    new_row = blank_row
    new_col = blank_col
    
    if dir == 1:    # RIGHT
        new_col += 1
    elif dir == 2:  # UP
        new_row -= 1
    elif dir == 3:  # LEFT
        new_col -= 1
    elif dir == 4:  # DOWN
        new_row += 1
        
    new_state = [list(r) for r in state]
    
    number_to_move = new_state[new_row][new_col]
    new_state[blank_row][blank_col] = number_to_move
    new_state[new_row][new_col] = 0
    
    return new_state

def h_misplaced(state, g_state):
    """
    Counts how many tiles are NOT in their correct goal position.
    We ignore the blank tile (0).
    """
    wrong_places = 0
    
    # Check every single square on the 3x3 grid
    for row in range(3):
        for col in range(3):
            current_tile = state[row][col]
            goal_tile = g_state[row][col]
            
            # We only care about actual numbered tiles, so skip the 0
            if current_tile != 0:
                # If the tile doesn't match the goal board, count it as misplaced
                if current_tile != goal_tile:
                    wrong_places += 1
                    
    return wrong_places

def h_manhattan(state, g_state):
    """
    Calculates the total 'walking distance' for all tiles to reach their goal.
    Distance = absolute difference in rows + absolute difference in columns.
    """
    total_distance = 0
    
    # Step 1: Look at every tile on our current board
    for row in range(3):
        for col in range(3):
            current_tile = state[row][col]
            
            # Skip the blank space (0)
            if current_tile == 0:
                continue
                
            # Step 2: Now, go find where this specific tile lives on the goal board
            target_row = -1
            target_col = -1
            
            for g_row in range(3):
                for g_col in range(3):
                    if g_state[g_row][g_col] == current_tile:
                        target_row = g_row
                        target_col = g_col
            
            # Step 3: Calculate the distance
            # The abs() function gives the absolute value (turns negatives into positives)
            # because you can't walk "negative" 2 steps!
            row_distance = abs(row - target_row)
            col_distance = abs(col - target_col)
            
            # Add this tile's distance to our total
            total_distance += (row_distance + col_distance)
            
    return total_distance

class Node:
    def __init__(self, state, parent=None, prev_movement=None, movements=None, level=0, g_cost=0, h_cost=0):
        self.state = state
        self.parent = parent
        self.immediate_children = []
        self.prev_movement = prev_movement
        self.movements = movements if movements is not None else []
        self.level = level
        self.g_cost = g_cost
        self.h_cost = h_cost
        self.f_cost = self.g_cost + self.h_cost
        self.key = self.state_to_key(state)

    @staticmethod
    def state_to_key(state):
        # Example: [[7,2,4],[5,0,6],[8,3,1]] -> 724506831
        return int("".join(str(tile) for row in state for tile in row))

    def update_costs(self, g_cost, h_cost):
        self.g_cost = g_cost
        self.h_cost = h_cost
        self.f_cost = g_cost + h_cost

def a_star_search(start_state, goal_state, heuristic_fn):
    """
    A* search for the 8-puzzle.

    Parameters:
        start_state : 3x3 list
        goal_state  : 3x3 list
        heuristic_fn(state, goal_state) -> int

    Returns:
        goal_node if solved, otherwise None
    """

    start_h = heuristic_fn(start_state, goal_state)
    start_node = Node(
        state=start_state,
        parent=None,
        prev_movement=None,
        movements=[],
        level=0,
        g_cost=0,
        h_cost=start_h
    )

    # Priority queue entries: (f_cost, h_cost, key, node)
    open_heap = []
    heapq.heappush(open_heap, (start_node.f_cost, start_node.h_cost, start_node.key, start_node))

    # # Best known g_cost for each state key
    # best_g_cost = {start_node.key: 0}

    # Closed set for already expanded states
    closed_set = set()

    while open_heap:
        _, _, _, current_node = heapq.heappop(open_heap)

        # Skip if already processed
        if current_node.key in closed_set:
            continue

        # Goal test
        if current_node.state == goal_state:
            return current_node

        closed_set.add(current_node.key)

        possible_moves = check_movement(current_node.state)

        for move in possible_moves:
            if move == 0:

            child_key = Node.state_to_key(child_state)

Important

OPEN ← priority queue, ordered by f(n) = g(n) + h(n)

CLOSED ← set of already-expanded states

  1. Add start to OPEN with f = h(start)
  2. While OPEN is not empty:
    1. Pop node n with lowest f from OPEN
    2. If n is GOAL → return solution path
    3. Add n to CLOSED
    4. For each neighbour m of n: new_g = g(n) + 1 If m not in CLOSED (or we found a cheaper path): Push m onto OPEN with f = new_g + h(m)
  3. Return failure (no solution)

heapq in Python implements a min-heap — heappop always returns the item with the smallest value, which is exactly the lowest-f node we want.

def print_tree(node, prefix="", is_last=True):

    if prefix == "":
        print(node.key)
    else:
        connector = "└── " if is_last else "├── "
        print(prefix + connector + "L" + str(node.level) + " K:" + str(node.key))

    child_count = len(node.immediate_children)

    for i, child in enumerate(node.immediate_children):
        last_child = (i == child_count - 1)

        if prefix == "":
            new_prefix = "    "
        else:
            new_prefix = prefix + ("    " if is_last else "│   ")

        print_tree(child, new_prefix, last_child)
